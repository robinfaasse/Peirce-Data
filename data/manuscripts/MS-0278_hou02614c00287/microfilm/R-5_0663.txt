$y_1 = y$
$x_1 = x + by$

$\begin{aligned}
x_1^2 + y_1^2 & = x^2 + 2bxy + (1+b^2)y^2 \\
\frac{x_1^2 + y_1^2}{x^2 + y^2} & = \frac{1 + 2b\frac{y}{x} + (1+b^2)(\frac{y}{x})^2}{1 + (\frac{y}{x})^2} \\
& = \frac{1 + 2b\tan\theta + (1+b^2)\tan^2\theta}{\sec^2\theta} \\
& = \cos^2\theta + b\sin 2\theta + (1+b^2)\sin^2\theta \\
& = 1 + b\sin 2\theta + b^2\sin^2\theta
\end{aligned}$

$(1 + b\sin 2\theta + b^2\sin^2\theta) = 2b\cos 2\theta + b^2\sin 2\theta$

$2\cos 2\theta + b\sin 2\theta = 0$
$\tan 2\theta = -\frac{2}{b} = \frac{2\tan\theta}{1-\tan^2\theta}$
$b = \tan\theta - \cot\theta$

& = \frac{\tan\alpha + \tan\beta}{1 - \tan\alpha\tan\beta}
\end{aligned}$
