[u]Midpoints of the edges[/u]

[marginalia]
1[sup]st[/sup] Triad
[/marginalia]
[fig]bracket grouping AB, EF, IJ[/fig]
AB = z
EF = x
IJ = y

CD = -z
GH = -x
KL = -y

[marginalia]
2[sup]nd[/sup] Triad
[/marginalia]
[fig]bracket grouping AE, AF, BG, BH[/fig]
AE = ((1+e)/2) x + e/2 y + 1/2 z
AF = ((1+e)/2) x - e/2 y + 1/2 z
BG = (-1-e)/2 x + e/2 y + 1/2 z
BH = (-1-e)/2 x - e/2 y + 1/2 z

DH = (-1-e)/2 x - e/2 y - 1/2 z
DG = (-1-e)/2 x + e/2 y - 1/2 z
CF = ((1+e)/2) x - e/2 y - 1/2 z
CE = ((1+e)/2) x + e/2 y - 1/2 z
