$$(\phi\phi)^{N+N} = \phi^N\phi^N = \phi^{N+N}$$
$$(\phi\phi)^{N+N} = (\phi\phi)\cdot(\phi\phi)^N$$
$$= (\phi\phi)\cdot(\phi^N\phi^N)$$
$$= \phi^N[\phi(\phi\phi^N)]$$
$$= [\phi(\phi\phi^N)]\phi^N$$
$$= [\phi\cdot\phi^{1+N}]\phi^N$$
$$= (\phi^{1+N}\cdot\phi)\phi^N$$
$$= \phi^{1+N}\cdot(\phi\cdot\phi^N)$$
$$= \phi^{1+N}\cdot\phi^{1+N}$$
