$$\frac{1}{R-N+x} = \frac{1}{R-N} \frac{1}{1+\frac{x}{R-N}} = \frac{1}{R-N} \left( 1 - \frac{x}{R-N} + \frac{x^2}{(R-N)^2} \right)$$
$$= \frac{1}{R-N} - \frac{x}{(R-N)^2} + \frac{x^2}{(R-N)^3}$$

[del]$$\frac{-a'}{R-N} + \frac{a'x}{(R-N)^2}$$[/del]
$$e = \frac{(2R-N)N}{2r} \left\{ \frac{1}{R-N} - \frac{x}{(R-N)^2} + \frac{x^2}{(R-N)^3} \right\} - \frac{a'}{R-N} + \frac{a'x}{(R-N)^2} + \frac{b'N}{R-N} - \frac{b'xN}{(R-N)^2}$$
$$- c \frac{I^2+2N+N^2}{R-N} + cx \frac{I^2+2N+N^2}{(R-N)^2} + c' \frac{I^2+2N}{R-N} - c'x \frac{I^2+2N}{(R-N)^2}$$

$$e^2 = \frac{(2R-N)^2 N^2}{4r^2} \left\{ \frac{1}{(R-N)^2} - \frac{2x}{(R-N)^3} + \frac{3x^2}{(R-N)^4} \right\} - \frac{(2R-N)N}{r} \left\{ \frac{a'}{(R-N)^2} - [del]\frac{a'x}{(R-N)^3}[/del] \right\}$$
$$+ \frac{a'^2}{(R-N)^2} + \frac{b'^2 N^2}{(R-N)^2} + c^2 \frac{(I^2+2N+N^2)^2}{(R-N)^2} + \frac{(2R-N)N}{r} \frac{a'x}{(R-N)^3}$$
$$+ c'^2 \frac{(I^2+2N)^2}{(R-N)^2} + \frac{(2R-N)N}{r} \left( \frac{b'N}{(R-N)^2} - 2 \frac{b'Nx}{(R-N)^3} \right)$$
$$+ \frac{(2R-N)N}{r} \left\{ -c \frac{I^2+2N+N^2}{(R-N)^2} + 2cx \frac{I^2+2N+N^2}{(R-N)^3} \right\}$$
$$+ \frac{(2R-N)N}{r} \left\{ c' \frac{I^2+2N}{(R-N)^2} - 2c'x \frac{I^2+2N}{(R-N)^3} \right\}$$

$$= \frac{(2R-N)^2 N^2}{4r^2 (R-N)^2} \left\{ 1 - \frac{2x}{R-N} + \frac{3x^2}{(R-N)^2} \right\} + \frac{(2R-N)N}{r(R-N)^2} \left[ [del]\frac{a'}{R-N} - \frac{2a'x}{R-N}[/del] -a' + b' - c(I^2+2N+N^2) + c'(I^2+2N) \right]$$
$$\left[ 1 - 2 \frac{x}{R-N} \right]$$
$$+ \frac{a'^2 + b'^2 N^2 + c^2 (I^2+2N+N^2)^2 + c'^2 (I^2+2N)^2}{(R-N)^2}$$

$$\sqrt{a+x} = \sqrt{a} \sqrt{1+\frac{x}{a}} = \sqrt{a} \left( 1 + \frac{1}{2} \frac{x}{a} - \frac{1}{8} \frac{x^2}{a^2} \right) = \sqrt{a} + \frac{1}{2} \frac{x}{\sqrt{a}} - \frac{1}{8} \frac{x^2}{a \sqrt{a}}$$
