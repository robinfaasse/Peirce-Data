$\text{Lift} = \delta \theta_n \frac{\cos (\theta_{n+1} - \theta_n)}{\cos \theta_{n+1}}$

$\text{Elongation} = \delta \theta_n \frac{\sin \theta_n}{\cos \theta_{n+1}}$

$\delta \theta_n \frac{\cos (\theta_{n+1} - \theta_n)}{\cos \theta_{n+1}} + \delta \theta_{n+1} \frac{\cos (\theta_{n+2} - \theta_{n+1})}{\cos \theta_{n+2}} = 0$

$\delta \theta_{n+1} = - \delta \theta_n \frac{\cos (\theta_{n+1} - \theta_n) \cos \theta_{n+2}}{\cos (\theta_{n+2} - \theta_{n+1}) \cos \theta_{n+1}}$

$\text{Total elongation} = \delta \theta_n \left( \frac{\sin \theta_n}{\cos \theta_{n+1}} - \frac{\cos (\theta_{n+1} - \theta_n)}{\cos (\theta_{n+2} - \theta_{n+1})} \frac{\sin \theta_{n+1}}{\cos \theta_{n+1}} \right)$
$= - \frac{\sin (\theta_{n+1} - \theta_n)}{\cos (\theta_{n+2} - \theta_{n+1})}$

$\cos \theta_{n+2} (\sin \theta_n - \sin \theta_{n+1})$
$\cos \theta_{n+2} \cos \theta_{n+1} \sin \theta_n + \sin \theta_{n+2} \sin \theta_{n+1} \sin \theta_n$
