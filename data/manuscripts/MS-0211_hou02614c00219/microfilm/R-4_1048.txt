[u]Fn[/u] = [u]p[/u] . F([u]n[/u]-1) + (1-[u]p[/u]) F([u]n[/u]+1)

or
F([u]n[/u]-1) = (1/[u]p[/u]) [u]Fn[/u] - [(1-[u]p[/u])/[u]p[/u]] F([u]n[/u]+1)
[u]Fn[/u] = (1/[u]p[/u]) F([u]n[/u]+1) - [(1-[u]p[/u])/[u]p[/u]] F([u]n[/u]+2)
or (1/[u]p[/u]) [u]Fn[/u] = (1/[u]p[/u])² F([u]n[/u]+1) - [(1-[u]p[/u])/[u]p[/u]²] F([u]n[/u]+2)
